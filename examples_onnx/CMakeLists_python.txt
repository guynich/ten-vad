cmake_minimum_required(VERSION 3.12)
project(ten_vad_python)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Detect architecture
execute_process(
    COMMAND uname -m
    OUTPUT_VARIABLE SYSTEM_ARCH
    OUTPUT_STRIP_TRAILING_WHITESPACE
)

message(STATUS "Detected architecture: ${SYSTEM_ARCH}")

# Set ONNX Runtime path based on architecture
if(SYSTEM_ARCH STREQUAL "x86_64")
    set(DEFAULT_ORT_PATH "$ENV{HOME}/onnxruntime-linux-x64-1.22.0")
elseif(SYSTEM_ARCH STREQUAL "aarch64")
    set(DEFAULT_ORT_PATH "$ENV{HOME}/onnxruntime-linux-aarch64-1.22.0")
else()
    message(FATAL_ERROR "Unsupported architecture: ${SYSTEM_ARCH}")
endif()

# Allow override of ONNX Runtime path
if(NOT DEFINED ORT_ROOT)
    set(ORT_ROOT ${DEFAULT_ORT_PATH})
endif()

message(STATUS "Using ONNX Runtime path: ${ORT_ROOT}")

# Verify ONNX Runtime path
if(NOT EXISTS "${ORT_ROOT}/lib" OR NOT EXISTS "${ORT_ROOT}/include")
    message(FATAL_ERROR "Invalid ONNX Runtime path: ${ORT_ROOT}")
endif()

# Find Python and pybind11
find_package(Python COMPONENTS Interpreter Development REQUIRED)
find_package(pybind11 REQUIRED)

# TEN VAD source directories
get_filename_component(TEN_VAD_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../ ABSOLUTE)
set(TEN_VAD_SRC_DIR ${TEN_VAD_ROOT}/src)
set(TEN_VAD_INCLUDE_DIR ${TEN_VAD_ROOT}/include)

# Source files
set(TEN_VAD_SOURCES
    ${TEN_VAD_SRC_DIR}/ten_vad.cc
    ${TEN_VAD_SRC_DIR}/aed.cc
    ${TEN_VAD_SRC_DIR}/biquad.cc
    ${TEN_VAD_SRC_DIR}/fftw.c
    ${TEN_VAD_SRC_DIR}/fscvrt.cc
    ${TEN_VAD_SRC_DIR}/pitch_est.cc
    ${TEN_VAD_SRC_DIR}/stft.cc
)

# Create pybind11 module
pybind11_add_module(ten_vad_python
    ten_vad_python.cpp
    ${TEN_VAD_SOURCES}
)

# Include directories
target_include_directories(ten_vad_python PRIVATE
    ${TEN_VAD_INCLUDE_DIR}
    ${TEN_VAD_SRC_DIR}
    ${ORT_ROOT}/include
)

# Link directories and libraries
target_link_directories(ten_vad_python PRIVATE
    ${ORT_ROOT}/lib
)

target_link_libraries(ten_vad_python PRIVATE
    onnxruntime
)

# Compiler flags
target_compile_options(ten_vad_python PRIVATE
    -Wno-write-strings
    -Wno-unused-result
)

# Set RPATH for ONNX Runtime library
set_target_properties(ten_vad_python PROPERTIES
    INSTALL_RPATH "${ORT_ROOT}/lib"
    BUILD_WITH_INSTALL_RPATH TRUE
    INSTALL_RPATH_USE_LINK_PATH TRUE
)

# Python module properties
set_target_properties(ten_vad_python PROPERTIES
    CXX_VISIBILITY_PRESET "hidden"
    VISIBILITY_INLINES_HIDDEN TRUE
)

# Installation (optional)
install(TARGETS ten_vad_python DESTINATION .)
